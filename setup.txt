# FastAPI＋uvicorn稼働環境
python3 -m venv venv
source venv/bin/activate
pip install -r requirements.txt
uvicorn main:app --reload --host 0.0.0.0 --port 8000

# 仮想環境を終了
deactivate

# 仮想環境のディレクトリごと削除
rm -rf .venv

# 外部公開用のURL
ngrok http 8000

# データベースに接続
cd backend
sqlite3 offishare.db

# テーブル一覧を表示
.tables

# 各テーブルのデータを確認
SELECT * FROM users;
SELECT * FROM transactions;
SELECT * FROM messages;

# DBから出る
.exit
